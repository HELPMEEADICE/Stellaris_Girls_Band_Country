# ==================== 薇欧拉病毒（violavirus）：完美假面网络协议 ====================
namespace = bgbn_violavirus


# 开局满 10 年后引出危机（引导事件，hide_window 不显示）
country_event = {
    id = bgbn_violavirus.1
    hide_window = yes
    is_triggered_only = yes

    trigger = {
        has_country_flag = poppinparty
    }

    immediate = {
        country_event = {
            id = bgbn_violavirus.2
            days = 3600
        }
    }
}

# 触发事件：开启局势
country_event = {
    id = bgbn_violavirus.2
    title = bgbn_violavirus.2.name
    desc = bgbn_violavirus.2.desc
    picture = GFX_evt_board_meeting
    is_triggered_only = yes

    trigger = {
        has_country_flag = poppinparty
    }

    fire_only_once = yes

    option = {
        name = bgbn_violavirus.2.a
        add_modifier = {
            modifier = bgbn_violavirus_1_a
            days = -1
        }
        if = {
            limit = { NOT = { country_has_situation = { SITUATION = situation_violavirus } } }
            start_situation = { type = situation_violavirus }
        }
    }
    option = {
        name = bgbn_violavirus.2.b
        add_resource = {
			energy = 250
			unity = 150
        }
        if = {
            limit = { NOT = { country_has_situation = { SITUATION = situation_violavirus } } }
            start_situation = { type = situation_violavirus }
        }
    }
}
