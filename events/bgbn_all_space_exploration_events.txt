	id = bgbn_space_exploration.1

    is_triggered_only = yes

	hide_window = yes

	

	trigger = {
		owner = {
			has_country_flag = poppinparty
			NOT = {
				has_country_flag = bgbn_space_exploration_second
			}
		}
	}
	immediate = {
		if = {
			limit = { 
				owner = {
					has_country_flag = bgbn_space_exploration_frist 
				}
			}
			owner = {
				country_event = {
					id = bgbn_space_exploration.2
					days = 1
				}
				set_country_flag = bgbn_space_exploration_second
			}
		}
		if = {
			limit = { 
				NOT = {
					owner = {
						has_country_flag = bgbn_space_exploration_frist
					}
				}
			}
			owner = {
				set_country_flag = bgbn_space_exploration_frist
			}
		}
	}
}
