namespace = ksm
namespace = bgbn_start

#设置country flag为poppinparty,后续各种内容将绑定这个flag,同时为了适配更多人的需求，选中闪耀谐律政体即可获得该flag,方便自建预设使用
country_event = {
	id = bgbn_start.1
	is_triggered_only = yes
	hide_window = yes
	trigger = {
		has_authority = auth_radiant_harmony
	}
	immediate = {
		owner = {
			set_country_flag = poppinparty
		}
	}
}

country_event = {
	id = ksm.0
	title = ksm.0.t
	desc = ksm.0.d
	#hide_window = yes
	is_triggered_only = yes
	picture = GFX_evt_kirakira
	trigger = {
		is_ai = no
	}
	immediate = {
		play_sound = { sound = tokimeki_experience location = capital_scope }	
		if = {
			limit = {
				has_country_flag = poppinparty
			}
			set_country_flag = ruled_by_ksm
			kill_leader = { 
				type = ruler 
				show_notification = no
			}
			create_species = {
				name = "BanGDream_Human"
				class = HUM
				portrait = "human"
				gender = female
				immortal = yes
				effect = {
					save_event_target_as = bangdream_species
				}
			}
		}
	}
	option = {
		name = ksm.0.a
		custom_tooltip = ksm_startup
		hidden_effect = {
			create_ksm = yes
			#ars
			create_ars = yes
			#saaya
			create_saaya = yes
			#tae
			create_tae = yes
			#rimi
			create_rimi = yes
			#asuka
			create_asuka = yes
		}
	}
}


country_event = {
	id = ksm.1
	hide_window = yes
	is_triggered_only = yes
	trigger = {
		is_ai = no
	}
	immediate = {
		if = {
			limit = {
				has_country_flag = poppinparty
			}
			every_owned_leader = {
            	limit = {
            	    NOT = {
            	        has_leader_flag = legendary_leader
            	    }
            	    NOT = { is_ruler = yes }
            	}
				remove_leader = THIS
            	kill_leader = { 
					type = THIS 
					show_notification = no
				}
        	}
		}
	}
}
