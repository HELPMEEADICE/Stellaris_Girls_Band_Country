country_event = {
	id = ksm.1
	hide_window = yes
	is_triggered_only = yes
	trigger = {
		is_ai = no
	}
	immediate = {
		if = {
			limit = {
				has_country_flag = poppinparty
			}
			every_owned_leader = {
            	limit = {
            	    NOT = {
            	        has_leader_flag = legendary_leader
            	    }
            	    NOT = { is_ruler = yes }
            	}
				remove_leader = THIS
            	kill_leader = { 
					type = THIS 
					show_notification = no
				}
        	}
			country_event = {
				id = ksm.0
			}
		}
	}
}
