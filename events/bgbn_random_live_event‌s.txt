country_event = {
	id = bgbn_ramdom_live.2
	title = bgbn_ramdom_live.2.name
	desc = bgbn_ramdom_live.2.desc
	picture = GFX_evt_star_yellow
	location = bgbn_random_live_2_planet
    location = bgbn_random_live_2_leader
	is_triggered_only = yes

	trigger = {
        has_country_flag = poppinparty
        
	}

	fire_only_once = yes

    immediate = {
        while = {   #循环
            count = 1   #循环一次
            random_owned_planet = { #随机一个拥有的行星，只需要随机一个的话不用写while
                limit = {
                    exists = leader
                }
                save_event_target_as = bgbn_random_live_2_planet
                leader = {
                    save_event_target_as = bgbn_random_live_2_leader
                }
            }
        }
    }

	option = {
		name = bgbn_ramdom_live.2.a
        event_target:bgbn_random_live_2_planet = {
            leader = {
                add_trait = { trait = leader_trait_bgbn_simple_resonator }
            }
        }
	}
}
