namespace = bgbn_voyage

country_event = {
    id = bgbn_voyage.1
    title = bgbn_voyage.1.name
    desc = bgbn_voyage.1.desc
    is_triggered_only = yes

    trigger = {
        has_country_flag = poppinparty
		if = {
			limit = {
				OR = {
					has_technology = tech_jump_drive_1
					has_technology = tech_psi_jump_drive_1
		        }
            }
	    }
	}
    
	fire_only_once = yes #只触发一次

    option = {
        name = bgbn_voyage.1.a
		owner = {
			country_event = { id = bgbn_voyage.2 days = 30}
		}	
    }
}
