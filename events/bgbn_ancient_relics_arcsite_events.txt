#星流  #触发考古1 星流
country_event = {				
	id = bgbn_ancrel.6          
	title = "bgbn_ancrel.6.name"
	desc = "bgbn_ancrel.6.desc"

	location = event_target:starry_string_planet		#注册特殊地点

	is_triggered_only = yes

	trigger = {
		has_authority = auth_radiant_harmony
	}

	immediate = {
		random_planet_within_border = {								#选中特殊地点
			limit = {
				OR = {
					is_planet_class = pc_barren_cold
					is_planet_class = pc_frozen
				}
				NOT = { exists = archaeological_site }				#选中行星的要求：没有考古遗迹
				NOT = { has_planet_flag = suppress_archaeological_sites }		#选中行星的要求：没有禁止考古遗迹生成
			}
			save_event_target_as = starry_string_planet
		}
	}

	option = {
		name = bgbn_ancrel.6.a
		event_target:starry_string_planet = {						#特殊地点会发生的事
			create_archaeological_site = site_starry_string
		}
	}
}

country_event = {						#生成 曦击 星系
	id = bgbn_ancrel.17        
	title = bgbn_ancrel.17.name
	desc = bgbn_ancrel.17.desc
	is_triggered_only = yes

	immediate = {
		set_country_flag = glowstrike_new_version
		set_country_flag = glowstrike_wand_new_version
		set_country_flag = glowstrike_version_11111
		
		set_spawn_system_batch = begin

		capital_scope = {
			spawn_system = {
				min_distance = 25
				max_distance = 30
				max_jumps = 0
				hyperlane = yes
				is_discovered = no
				initializer = "glowstrike_system"
				effect = {
					save_global_event_target_as = glowstrike_system
				}
			}
		}

		set_spawn_system_batch = end
	}


	option = { 
		name = bgbn_ancrel.17.a
		owner = {
			country_event = { id = bgbn_ancrel.18 days = 1 }
		}
	}
}
