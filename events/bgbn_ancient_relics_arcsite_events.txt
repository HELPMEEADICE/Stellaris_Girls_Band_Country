#星流
country_event = {				#触发考古的事件是country_event,考古内部的阶段事件是fleet_event
	id = bgbn_ancrel.0          #触发考古1 星流
	title = "bgbn_ancrel.0.name"
	desc = "bgbn_ancrel.0.desc"

	is_triggered_only = yes

	trigger = {
		has_authority = auth_radiant_harmony
		days_passed >= 730
    }

	mean_time_to_happen = {
		days = 0
	}

	fire_only_once = yes

	immediate = {
		random_planet_within_border = {
			limit = {
				OR = {
					is_planet_class = pc_molten
					is_planet_class = pc_barren
					is_planet_class = pc_barren_cold
					is_planet_class = pc_frozen
				}
				NOT = { exists = archaeological_site }
				NOT = { has_planet_flag = suppress_archaeological_sites }
			}
			save_event_target_as = tomb_planet
		}
	}

	option = {
		name = bgbn_ancrel.0.a
		event_target:tomb_planet = {
			create_archaeological_site = site_starry_string
		}
	}
}
