#星流  #触发考古1 星流
country_event = {				
	id = bgbn_ancrel.6          
	title = "bgbn_ancrel.6.name"
	desc = "bgbn_ancrel.6.desc"

	location = event_target:starry_string_planet		#注册特殊地点

	is_triggered_only = yes

	trigger = {
		has_country_flag = poppinparty
	}

	immediate = {
		random_planet_within_border = {								#选中特殊地点
			limit = {
				OR = {
					is_planet_class = pc_barren_cold
					is_planet_class = pc_frozen
				}
				NOT = { exists = archaeological_site }				#选中行星的要求：没有考古遗迹
				NOT = { has_planet_flag = suppress_archaeological_sites }		#选中行星的要求：没有禁止考古遗迹生成
			}
			save_event_target_as = starry_string_planet
		}
	}

	option = {
		name = bgbn_ancrel.6.a
		event_target:starry_string_planet = {						#特殊地点会发生的事
			create_archaeological_site = site_starry_string
		}
	}
}
