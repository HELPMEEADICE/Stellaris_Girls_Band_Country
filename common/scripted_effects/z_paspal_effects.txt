create_aya = {
	create_leader = {
		species = event_target:bangdream_species
		class = official										#领袖类型
		skip_background_generation = no							#no使领袖生成母星，思潮，此前岗位 yes跳过
		name = PRESCRIPTED_ruler_name_aya1
		custom_description = PRESCRIPTED_ruler_name_aya1_desc
		immortal = yes
		set_age = 18
		hide_age = yes
		gender = female
		skill = 10
		randomize_traits = no
		background_ethic = ethic_xenophile						#设定领袖思潮
		traits = {
			trait = leader_trait_star_beat
			trait = trait_paspal
		}
		event_leader = yes
		effect = {
			change_leader_portrait = aya_leader_paspal_portrait
			set_leader_flag = aya
			set_leader_flag = paspal
			set_leader_flag = legendary_leader
			change_background_job = bgbn_vocal					#设定领袖此前岗位
		}
	}
}
