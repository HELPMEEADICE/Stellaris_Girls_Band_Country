		custom_description = PRESCRIPTED_ruler_name_arale1_desc

		set_age = 18
		hide_age = yes
		gender = female
		skill = 10
		randomize_traits = no
		background_ethic = ethic_xenophile

			trait = leader_trait_star_beat
			trait = trait_mugendai
		}
