create_raika = { #须贺蕾叶
	create_leader = {
		species = event_target:bangdream_species
		class = commander
		skip_background_generation = no
		name = PRESCRIPTED_ruler_name_raika1
		custom_description = PRESCRIPTED_ruler_name_raika1_desc
		immortal = yes
		set_age = 18
		hide_age = yes
		gender = female
		skill = 10
		randomize_traits = no
		background_ethic = ethic_militarist
		traits = {
			trait = leader_trait_star_beat
			trait = trait_dumbrock
		}
		event_leader = yes
		effect = {
			change_leader_portrait = raika_leader_dumbrock_portrait
			set_leader_flag = raika
			set_leader_flag = dumbrock
			set_leader_flag = legendary_leader
			change_background_job = bgbn_vocal
			save_global_event_target_as = raika #方便在事件中锁定领袖
		}
	}
}
