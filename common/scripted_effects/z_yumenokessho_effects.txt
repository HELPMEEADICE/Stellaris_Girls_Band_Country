create_rose = {
	create_leader = {
		species = event_target:bangdream_species
		class = commander
		skip_background_generation = no
		name = PRESCRIPTED_ruler_name_rose1
		custom_description = PRESCRIPTED_ruler_name_rose1_desc
		immortal = yes
		set_age = 18
		hide_age = yes
		gender = female
		skill = 10
		randomize_traits = no
		background_ethic = ethic_militarist
		traits = {
			trait = leader_trait_star_beat
			trait = trait_yumenokessho
		}
		event_leader = yes
		effect = {
			change_leader_portrait = rose_leader_yumenokessho_portrait
			set_leader_flag = rose
			set_leader_flag = yumenokessho
			set_leader_flag = legendary_leader
			change_background_job = bgbn_yumenokessho
			save_global_event_target_as = rose #方便在事件中锁定领袖
		}
	}
}
