create_chuchu = {
	create_leader = {
		species = event_target:bangdream_species
		class = commander
		skip_background_generation = no
		name = PRESCRIPTED_ruler_name_chuchu1
		custom_description = PRESCRIPTED_ruler_name_chuchu1_desc
		immortal = yes
		set_age = 18
		hide_age = yes
		gender = female
		skill = 10
		randomize_traits = no
		background_ethic = ethic_authoritarian 
		traits = {
			trait = leader_trait_star_beat
			trait = trait_ras
		}
		event_leader = yes
		effect = {
			change_leader_portrait = chuchu_leader_ras_portrait
			set_leader_flag = chuchu
			set_leader_flag = ras
			set_leader_flag = legendary_leader
			change_background_job = bgbn_dj
		}
	}
}
