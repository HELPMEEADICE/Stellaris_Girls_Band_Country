create_tomori = { #高松灯
	create_leader = {
		species = event_target:bangdream_species
		class = scientist
		skip_background_generation = no
		name = PRESCRIPTED_ruler_name_tomori1
		custom_description = PRESCRIPTED_ruler_name_tomori1_desc
		immortal = yes
		set_age = 18
		hide_age = yes
		gender = female
		skill = 10
		randomize_traits = no
		background_ethic = ethic_spiritualist
		traits = {
			trait = leader_trait_star_beat
			trait = trait_mygo
		}
		event_leader = yes
		effect = {
			change_leader_portrait = tomori_leader_mygo_portrait
			set_leader_flag = tomori
			set_leader_flag = mygo
			set_leader_flag = legendary_leader
			change_background_job = bgbn_vocal
			save_global_event_target_as = tomori
		}
	}
}
