create_mashiro = {	#仓田真白
	create_leader = {
		species = event_target:bangdream_species
		class = scientist
		skip_background_generation = no
		name = PRESCRIPTED_ruler_name_mashiro1
		custom_description = PRESCRIPTED_ruler_name_mashiro1_desc
		immortal = yes
		set_age = 18
		hide_age = yes
		gender = female
		skill = 10
		randomize_traits = no
		background_ethic = ethic_spiritualist
		traits = {
			trait = leader_trait_star_beat
			trait = trait_monica
		}
		event_leader = yes
		effect = {
			change_leader_portrait = mashiro_leader_monica_portrait
			set_leader_flag = mashiro
			set_leader_flag = monica
			set_leader_flag = legendary_leader
			change_background_job = bgbn_vocal
			save_global_event_target_as = mashiro
		}
	}
}
