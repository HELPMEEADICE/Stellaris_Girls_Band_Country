	create_leader = {
		species = event_target:bangdream_species
		class = commander
		skip_background_generation = no
		name = PRESCRIPTED_ruler_name_ykn1
		custom_description = PRESCRIPTED_ruler_name_ykn1_desc
		immortal = yes
		set_age = 18
		hide_age = yes
		gender = female
		skill = 10
		randomize_traits = no
		background_ethic = ethic_militarist
		traits = {
			trait = leader_trait_star_beat
			trait = trait_roselia
		}
		event_leader = yes
		effect = {
			change_leader_portrait = ykn_leader_roselia_portrait
			set_leader_flag = ykn
			set_leader_flag = roselia
			set_leader_flag = legendary_leader
			change_background_job = bgbn_vocal
		}
	}
}
