create_sakiko = { #丰川祥子
	create_leader = {
		species = event_target:bangdream_species
		class = official
		skip_background_generation = no
		name = PRESCRIPTED_ruler_name_sakiko1
		custom_description = PRESCRIPTED_ruler_name_sakiko1_desc
		immortal = yes
		set_age = 18
		hide_age = yes
		gender = female
		skill = 10
		randomize_traits = no
		background_ethic = ethic_authoritarian
		traits = {
			trait = leader_trait_star_beat
			trait = trait_mujica
		}
		event_leader = yes
		effect = {
			change_leader_portrait = sakiko_leader_mujica_portrait
			set_leader_flag = sakiko
			set_leader_flag = mujica
			set_leader_flag = legendary_leader
			change_background_job = bgbn_keyboardist
			save_global_event_target_as = sakiko
		}
	}
}
