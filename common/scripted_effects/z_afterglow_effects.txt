create_ran = {
	create_leader = {
		species = event_target:bangdream_species
		class = commander
		skip_background_generation = no
		name = PRESCRIPTED_ruler_name_ran1
		custom_description = PRESCRIPTED_ruler_name_ran1_desc
		immortal = yes
		set_age = 18
		hide_age = yes
		gender = female
		skill = 10
		randomize_traits = no
		background_ethic = ethic_pacifist
		traits = {
			trait = leader_trait_star_beat
			trait = trait_ag
		}
		event_leader = yes
		effect = {
			change_leader_portrait = ran_leader_ag_portrait
			set_leader_flag = ran
			set_leader_flag = ag
			set_leader_flag = legendary_leader
			change_background_job = bgbn_vocal
			save_global_event_target_as = ran
		}
	}
}
