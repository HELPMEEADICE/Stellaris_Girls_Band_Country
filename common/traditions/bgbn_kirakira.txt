#传统内容文件
tr_kirakira_adopt = {	#采纳效果
	modifier = {
		BIOLOGICAL_species_trait_points_add = 2   	
		BIOLOGICAL_species_trait_picks_add = 2
	}
	custom_tooltip_with_modifiers  = tr_kirakira_adopt_tooltip	#效果提示
	on_enabled = {
		hidden_effect = {
			give_technology = {
				tech = tech_gene_tailoring
				message = yes
			}
			species = {
				change_species_characteristics = {
					add_trait = trait_first_glowing
					add_traits_at_start_of_list = yes
				}
			}
			every_owned_species = {
				limit = {
					is_subspecies = root
				}
				change_species_characteristics = {
					add_trait = trait_first_glowing
					add_traits_at_start_of_list = yes
				}
			}
		}
	}
}

tr_kirakira_finish = {	#传统完成效果
	modifier = {
		species_leader_exp_gain = 1	#领袖经验获得
		logistic_growth_mult = 0.2            #人口增长速度
		ascension_perks_add = 2		#飞升槽位解锁
	}
}

tr_kirakira_socialized_upbringing = {	#1
	modifier = {
		logistic_growth_mult = 3	 #人口增长速度
		bonus_pop_growth = 3		#每月有机人口生长
		pop_demotion_time_mult = -1.5	
		BIOLOGICAL_species_trait_points_add = 5   	#有机物种特质修饰点
		BIOLOGICAL_species_trait_picks_add = 3		#可选有机物种特质数
	}
	ai_weight = {
		factor = 1000
	}
}

tr_kirakira_post_scarcity_production_paradigm = { #2
	modifier = {
		empire_size_pops_mult = -0.75	#人口帝国规模
		planet_jobs_produces_mult = 1     #所有岗位产出
		planet_pops_energy_upkeep_mult = -0.9	#人口岗位能量币维护
		planet_pops_consumer_goods_upkeep_mult = -0.9 #人口岗位消费品维护
		pop_amenities_usage_mult = -0.5	#人口舒适度使用
	}
	ai_weight = {
		factor = 1000
	}
}

tr_kirakira_cognitive_extension = {	#3
	possible = {
		has_tradition = tr_kirakira_socialized_upbringing
		has_tradition = tr_kirakira_post_scarcity_production_paradigm
	}
	modifier = {
		BIOLOGICAL_species_trait_points_add = 5   	#有机物种特质修饰点
		pop_environment_tolerance = 1	#宜居性
		science_ship_survey_speed = 1 #调查速度
		ship_emergency_ftl_mult = -1 #舰船超光速脱离风险
		#允许经由物种修饰技术移除有益特质，增加不利特质。允许经由物种修饰技术调整物种外观。可以同化蜂巢思维    可以在game_rule和scripted_triggers里面绑定效果，选择has_这个传统
	}
	ai_weight = {
		factor = 1000
	}
	custom_tooltip_with_modifiers = tr_kirakira_cognitive_extensio_desc
}
