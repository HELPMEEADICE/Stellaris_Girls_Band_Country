### glowstrike System 曦击 星系
glowstrike_system = {
	name = "glowstrike.name"
	class = "sc_neutron_star"							#星系类型
	flags = { glowstrike_system_flag }

	planet = {
		name = "glowstrike_system.1.name"
		class = "pc_neutron_star"						#行星/恒星类型
		flags = { glowstrike_star_flag }				#可以用于各种事件/效果锁定星球
		orbit_distance = 0								#轨道距离 0代表中心
		size = { min = 37 max = 71 }
		init_effect = {
		    prevent_anomaly = yes
		    clear_deposits = yes
		    add_deposit = d_physics_30
			set_planet_flag = glowstrike_star
			save_global_event_target_as = glowstrike_star
		}	
	}
}
